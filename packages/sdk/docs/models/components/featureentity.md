# FeatureEntity

## Example Usage

```typescript
import { FeatureEntity } from "creem/models/components";

let value: FeatureEntity = {
  id: "feat_abc123",
  type: "customerCredits",
  description: "Access to premium course materials.",
  externalId: "pro-credits",
  customerCredits: {
    amount: "100",
    unitLabel: "tokens",
    bucketName: "images",
    requiresModeration: false,
    resetsEachPeriod: false,
  },
  licenseKey: {
    activationLimit: 3,
    expiry: {
      interval: "month",
      intervalCount: 1,
    },
  },
  privateNote: "Thanks for your purchase! Your access code is XYZ123.",
  file: {
    files: [],
  },
};
```
